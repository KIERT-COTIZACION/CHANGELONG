-- ============================================================
-- BASE DE DATOS: CotizacionesDB
-- Sistema de Cotización con Inventario
-- ============================================================

-- 1. Crear la base de datos
IF NOT EXISTS (SELECT * FROM sys.databases WHERE name = 'CotizacionesDB')
BEGIN
    CREATE DATABASE CotizacionesDB;
END
GO

USE CotizacionesDB;
GO

-- ============================================================
-- 2. TABLA: Usuarios
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Usuarios' AND xtype='U')
CREATE TABLE Usuarios (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    Username        NVARCHAR(50)  NOT NULL UNIQUE,
    PasswordHash    NVARCHAR(256) NOT NULL,
    NombreCompleto  NVARCHAR(150),
    Rol             NVARCHAR(20)  DEFAULT 'vendedor',
    Activo          BIT           DEFAULT 1,
    FechaCreacion   DATETIME      DEFAULT GETDATE()
);
GO

-- ============================================================
-- 3. TABLA: EmpresaConfig (logo + datos + colores)
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='EmpresaConfig' AND xtype='U')
CREATE TABLE EmpresaConfig (
    Id                    INT IDENTITY(1,1) PRIMARY KEY,
    NombreEmpresa         NVARCHAR(200) NOT NULL,
    Ruc                   NVARCHAR(20),
    Direccion             NVARCHAR(300),
    Telefono1             NVARCHAR(50),
    Telefono2             NVARCHAR(50),
    Email                 NVARCHAR(150),
    Instagram             NVARCHAR(100),
    Facebook              NVARCHAR(100),
    SitioWeb              NVARCHAR(150),
    LogoImage             VARBINARY(MAX),
    LogoNombreArchivo     NVARCHAR(200),
    DatosBancariosJson    NVARCHAR(MAX),
    NotasCotizacion       NVARCHAR(MAX),
    ValidezDias           INT           DEFAULT 30,
    TiempoEntrega         NVARCHAR(100) DEFAULT '5-10 días hábiles',
    PorcentajeIGV         DECIMAL(5,2)  DEFAULT 18.00,
    Moneda                NVARCHAR(10)  DEFAULT 'S/',
    ColorPrimario         NVARCHAR(20)  DEFAULT '#69383e',
    ColorSecundario       NVARCHAR(20)  DEFAULT '#a6a5a0',
    ColorTerciario        NVARCHAR(20)  DEFAULT '#5d5a55',
    ColorFondo            NVARCHAR(20)  DEFAULT '#fefefe',
    FechaModificacion     DATETIME      DEFAULT GETDATE()
);
GO

-- ============================================================
-- 4. TABLA: Clientes
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Clientes' AND xtype='U')
CREATE TABLE Clientes (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    Nombre          NVARCHAR(200) NOT NULL,
    Documento       NVARCHAR(50),
    Telefono        NVARCHAR(50),
    Email           NVARCHAR(150),
    Direccion       NVARCHAR(300),
    FechaCreacion   DATETIME DEFAULT GETDATE()
);
GO

-- ============================================================
-- 5. TABLA: Productos
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Productos' AND xtype='U')
CREATE TABLE Productos (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    Codigo          NVARCHAR(50) NOT NULL UNIQUE,
    Descripcion     NVARCHAR(200) NOT NULL,
    Precio          DECIMAL(18,2) DEFAULT 0,
    Stock           INT           DEFAULT 0,
    FechaCreacion   DATETIME      DEFAULT GETDATE()
);
GO

-- ============================================================
-- 6. TABLA: Inventario (tallas y colores por producto)
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Inventario' AND xtype='U')
CREATE TABLE Inventario (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    ProductoId      INT NOT NULL,
    Talla           NVARCHAR(50),
    Color           NVARCHAR(100),
    Cantidad        INT DEFAULT 0,
    FOREIGN KEY (ProductoId) REFERENCES Productos(Id) ON DELETE CASCADE
);
GO

-- ============================================================
-- 7. TABLA: Cotizaciones
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Cotizaciones' AND xtype='U')
CREATE TABLE Cotizaciones (
    Id                  INT IDENTITY(1,1) PRIMARY KEY,
    Numero              NVARCHAR(50) NOT NULL UNIQUE,
    Fecha               DATETIME     DEFAULT GETDATE(),
    ClienteId           INT,
    ClienteNombre       NVARCHAR(200),
    ClienteDocumento    NVARCHAR(50),
    Subtotal            DECIMAL(18,2) DEFAULT 0,
    IGV                 DECIMAL(18,2) DEFAULT 0,
    Total               DECIMAL(18,2) DEFAULT 0,
    ConIGV              BIT           DEFAULT 1,
    Observaciones       NVARCHAR(MAX),
    UsuarioId           INT,
    FOREIGN KEY (ClienteId) REFERENCES Clientes(Id),
    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

-- ============================================================
-- 8. TABLA: Detalle de Cotizaciones
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='CotizacionDetalles' AND xtype='U')
CREATE TABLE CotizacionDetalles (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    CotizacionId    INT NOT NULL,
    ProductoId      INT,
    Codigo          NVARCHAR(50),
    Descripcion     NVARCHAR(200),
    Precio          DECIMAL(18,2) DEFAULT 0,
    Cantidad        INT           DEFAULT 0,
    Subtotal        DECIMAL(18,2) DEFAULT 0,
    FOREIGN KEY (CotizacionId) REFERENCES Cotizaciones(Id) ON DELETE CASCADE
);
GO
