-- ============================================================
-- 2. TABLA: Usuarios
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Usuarios' AND xtype='U')

    Id              INT IDENTITY(1,1) PRIMARY KEY,

-- ============================================================
-- 5. TABLA: Clientes
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Clientes' AND xtype='U')
CREATE TABLE Clientes (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    Nombre          NVARCHAR(200) NOT NULL,
    Documento       NVARCHAR(50),
    Telefono        NVARCHAR(50),
    Email           NVARCHAR(150),
    Direccion       NVARCHAR(300),
    FechaCreacion   DATETIME DEFAULT GETDATE()
);
GO

-- ============================================================
-- 6. TABLA: Productos
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Productos' AND xtype='U')
CREATE TABLE Productos (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    Codigo          NVARCHAR(50) NOT NULL UNIQUE,
    Descripcion     NVARCHAR(200) NOT NULL,
    Precio          DECIMAL(18,2) DEFAULT 0,
    Stock           INT           DEFAULT 0,
    FechaCreacion   DATETIME      DEFAULT GETDATE()
);
GO

-- ============================================================
-- 7. TABLA: Inventario (tallas y colores por producto)
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Inventario' AND xtype='U')
CREATE TABLE Inventario (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    ProductoId      INT NOT NULL,
    Talla           NVARCHAR(50),
    Color           NVARCHAR(100),
    Cantidad        INT DEFAULT 0,
    FOREIGN KEY (ProductoId) REFERENCES Productos(Id) ON DELETE CASCADE
);
GO

-- ============================================================
-- 8. TABLA: Cotizaciones (cabecera)
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='Cotizaciones' AND xtype='U')
CREATE TABLE Cotizaciones (
    Id                  INT IDENTITY(1,1) PRIMARY KEY,
    Numero              NVARCHAR(50) NOT NULL UNIQUE,
    Fecha               DATETIME     DEFAULT GETDATE(),
    ClienteId           INT,
    ClienteNombre       NVARCHAR(200),
    ClienteDocumento    NVARCHAR(50),
    Subtotal            DECIMAL(18,2) DEFAULT 0,
    IGV                 DECIMAL(18,2) DEFAULT 0,
    Total               DECIMAL(18,2) DEFAULT 0,
    ConIGV              BIT           DEFAULT 1,
    Observaciones       NVARCHAR(MAX),
    UsuarioId           INT,
    FOREIGN KEY (ClienteId) REFERENCES Clientes(Id),
    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

-- ============================================================
-- 9. TABLA: CotizacionDetalles
-- ============================================================
IF NOT EXISTS (SELECT * FROM sysobjects WHERE name='CotizacionDetalles' AND xtype='U')
CREATE TABLE CotizacionDetalles (
    Id              INT IDENTITY(1,1) PRIMARY KEY,
    CotizacionId    INT NOT NULL,
    ProductoId      INT           NULL,
    Codigo          NVARCHAR(50)  NULL,
    Descripcion     NVARCHAR(500) NULL,
    Talla           NVARCHAR(50)  NULL,
    Color           NVARCHAR(100) NULL,
    Precio          DECIMAL(18,2) DEFAULT 0,
    Cantidad        INT           DEFAULT 0,
    Subtotal        DECIMAL(18,2) DEFAULT 0,
    EsLibre         BIT           DEFAULT 0,
    FOREIGN KEY (CotizacionId) REFERENCES Cotizaciones(Id) ON DELETE CASCADE
);
GO

-- ============================================================
-- 11. LIMPIAR DUPLICADOS (POR SI ACASO)
-- ============================================================
PRINT '--- Limpiando duplicados de Productos ---';
WITH Duplicados AS (
    SELECT Id,
           ROW_NUMBER() OVER (PARTITION BY Codigo ORDER BY Id) AS rn
    FROM Productos
)
DELETE FROM Productos
WHERE Id IN (SELECT Id FROM Duplicados WHERE rn > 1);
GO

PRINT '--- Limpiando duplicados de Inventario ---';
WITH DuplicadosInv AS (
    SELECT Id,
           ROW_NUMBER() OVER (PARTITION BY ProductoId, Talla, Color ORDER BY Id) AS rn
    FROM Inventario
)
DELETE FROM Inventario
WHERE Id IN (SELECT Id FROM DuplicadosInv WHERE rn > 1);
GO

-- 12. VERIFICACIÓN
-- ============================================================
PRINT '--- Verificación de tablas ---';
SELECT 'Usuarios' AS Tabla, COUNT(*) AS Registros FROM Usuarios
UNION ALL SELECT 'EmpresaConfig', COUNT(*) FROM EmpresaConfig
UNION ALL SELECT 'Bancos', COUNT(*) FROM Bancos
UNION ALL SELECT 'Clientes', COUNT(*) FROM Clientes
UNION ALL SELECT 'Productos', COUNT(*) FROM Productos
UNION ALL SELECT 'Inventario', COUNT(*) FROM Inventario
UNION ALL SELECT 'Cotizaciones', COUNT(*) FROM Cotizaciones
UNION ALL SELECT 'CotizacionDetalles', COUNT(*) FROM CotizacionDetalles;
GO

PRINT '--- Productos ---';
SELECT Id, Codigo, Descripcion, Precio, Stock 
FROM Productos ORDER BY Codigo;
GO

PRINT '--- Inventario con nombre de producto ---';
SELECT p.Codigo, p.Descripcion, i.Talla, i.Color, i.Cantidad
FROM Productos p
LEFT JOIN Inventario i ON i.ProductoId = p.Id
ORDER BY p.Codigo, i.Id;
GO

PRINT '--- Estructura de CotizacionDetalles ---';
SELECT COLUMN_NAME, DATA_TYPE, CHARACTER_MAXIMUM_LENGTH, IS_NULLABLE
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_NAME = 'CotizacionDetalles'
ORDER BY ORDINAL_POSITION;
GO

PRINT '--- Índices creados ---';
SELECT 
    t.name AS Tabla,
    i.name AS Indice,
    i.type_desc AS Tipo
FROM sys.indexes i
INNER JOIN sys.tables t ON i.object_id = t.object_id
WHERE i.name IS NOT NULL 
  AND t.name IN ('Cotizaciones', 'CotizacionDetalles', 'Inventario', 'Clientes', 'Productos')
ORDER BY t.name, i.name;
GO
